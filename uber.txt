# Define a function to calculate the geographical distance
# between pickup and dropoff locations using the Haversine formula
def haversine(lat1, lon1, lat2, lon2):

    # Convert latitude and longitude values from degrees to radians
    lat1, lon1, lat2, lon2 = map(
        np.radians,
        [lat1, lon1, lat2, lon2]
    )

    # Calculate the difference between latitudes
    dlat = lat2 - lat1

    # Calculate the difference between longitudes
    dlon = lon2 - lon1

    # Apply the Haversine formula
    a = (
        np.sin(dlat / 2) ** 2
        + np.cos(lat1) * np.cos(lat2)
        * np.sin(dlon / 2) ** 2
    )

    # Calculate and return the distance in kilometers
    # 6371 is the approximate radius of Earth in kilometers
    return 6371 * 2 * np.arcsin(np.sqrt(a))

# Remove invalid records using conditions

    # Distance must be greater than zero
    (df['distance_km'] > 0)
]


# Define a function to remove outliers using the IQR method
def remove_outliers(data, column):

    # Calculate the first quartile (25th percentile)
    Q1 = data[column].quantile(0.25)

    # Calculate the third quartile (75th percentile)
    Q3 = data[column].quantile(0.75)

    # Calculate the Interquartile Range
    IQR = Q3 - Q1

    # Calculate the lower limit for detecting outliers
    lower = Q1 - 1.5 * IQR

    # Calculate the upper limit for detecting outliers
    upper = Q3 + 1.5 * IQR

    # Return only rows whose values are within the limits
    return data[
        (data[column] >= lower) &
        (data[column] <= upper)
    ]

# Select input features that will be used for prediction
features = [
    'pickup_longitude',
    'pickup_latitude',
    'dropoff_longitude',
    'dropoff_latitude',
    'passenger_count',
    'distance_km',
    'hour',
    'day',
    'month',
    'weekday'
]

# Store the selected input features in X
X = df[features]

# Store fare_amount as the target variable in y
y = df['fare_amount']


# Split the dataset into training and testing sets
X_train, X_test, y_train, y_test = train_test_split(
    X,                   # Input features
    y,                   # Target variable
    test_size=0.2,       # 20% data for testing and 80% for training
    random_state=42      # Fix random state to get repeatable results
)

# Display the number of training samples
print("Training samples:", len(X_train))

# Display the number of testing samples
print("Testing samples:", len(X_test))


# Create a Linear Regression model
lr = LinearRegression()

# Train the Linear Regression model using training data
lr.fit(X_train, y_train)

# Predict fare amounts using the testing data
lr_pred = lr.predict(X_test)

# Calculate the R2 score of Linear Regression
lr_r2 = r2_score(y_test, lr_pred)

# Calculate Root Mean Squared Error
lr_rmse = np.sqrt(mean_squared_error(y_test, lr_pred))

# Calculate Mean Absolute Error
lr_mae = mean_absolute_error(y_test, lr_pred)

# Display Linear Regression model results
print("Linear Regression")
print("R2 :", lr_r2)
print("RMSE :", lr_rmse)
print("MAE :", lr_mae)


# Create a Random Forest Regression model

)

# Calculate the R2 score of Random Forest
rf_r2 = r2_score(y_test, rf_pred)

# Calculate Root Mean Squared Error

# Display Random Forest model results
